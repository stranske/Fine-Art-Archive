"""Execute offline semantic mutations and restore exact bytes after each control."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import xml.etree.ElementTree as ET

root = Path('/tmp/issue-772-round')
main = Path('src/fine_art_archive/api/main.py')
gates = Path('src/fine_art_archive/api/gates.py')
test = Path('tests/test_api_artist_decisions.py')
originals = {path: path.read_bytes() for path in (main, gates, test)}
sha = lambda data: hashlib.sha256(data).hexdigest()
prefix = str(test) + '::'
choice = prefix + 'test_artist_decision_persists_choice_and_reports_current_approvals'
sequence = prefix + 'test_later_artist_decision_reverses_choice_without_erasing_history'
invalid = prefix + 'test_invalid_artist_qid_is_rejected_before_log_append'
unsupported = prefix + 'test_unsupported_artist_decision_is_rejected_before_log_append'
main_text = originals[main].decode()
start = main_text.index('def artist_decision(')
end = main_text.index('\n\n@app.post(', start)
endpoint = main_text[start:end]
append_start = endpoint.index('    gates.append_allowlist(')
append_end = endpoint.index('    allow = ', append_start)
append_block = endpoint[append_start:append_end]
controls = [
    ('omit-append', main, 'endpoint', append_block, '', [choice, sequence]),
    ('wrong-artist', main, 'endpoint', '        artist_qid,', '        "Q43",', [choice, sequence]),
    ('reverse-choice', main, 'endpoint', 'decision=body.decision,', 'decision="reject" if body.decision == "approve" else "approve",', [choice, sequence]),
    ('discard-name', main, 'endpoint', 'artist_name=body.artist_name,', 'artist_name="",', [choice]),
    ('discard-note', main, 'endpoint', 'note=body.note,', 'note="",', [choice]),
    ('discard-reviewer', main, 'endpoint', 'reviewer=body.reviewer,', 'reviewer="tim",', [choice]),
    ('wrong-timestamp', main, 'endpoint', 'ts=datetime.now(UTC).isoformat(),', 'ts="2026-10-01T00:00:00+00:00",', [choice, sequence]),
    ('wrong-default-reviewer', main, 'endpoint', 'reviewer=body.reviewer,', 'reviewer=body.reviewer if body.reviewer != "tim" else "other",', [sequence]),
    ('wrong-approval-count', main, 'endpoint', '"approved_artists": len(allow),', '"approved_artists": len(allow) + 1,', [choice, sequence]),
    ('omit-qid-validation', main, 'endpoint', '    if not re.fullmatch(r"Q[1-9][0-9]{0,11}", artist_qid):\n        raise HTTPException(400, "artist_qid must look like Q12345")\n', '', [invalid]),
    ('accept-unsupported-choice', main, 'model', '    decision: Literal["approve", "reject"]', '    decision: Literal["approve", "reject", "defer"]', [unsupported]),
    ('overwrite-history', gates, 'append', 'with open(p, "a", encoding="utf-8") as fh:', 'with open(p, "w", encoding="utf-8") as fh:', [choice, sequence]),
    ('keep-rejected-approval', gates, 'reader', '            approved.discard(qid)', '            pass', [sequence]),
]
env = os.environ.copy()
env.update(PYTHONPATH=str(root / 'shim') + ':src', PYTHONDONTWRITEBYTECODE='1')

def purge_bytecode():
    for path in (main, gates):
        for compiled in (path.parent / '__pycache__').glob(path.stem + '.*.pyc'):
            compiled.unlink()

def run(name, phase, nodes):
    purge_bytecode()
    xml = root / f'{name}-{phase}.xml'
    command = [sys.executable, '-m', 'pytest', *nodes, '-m', 'not slow', '--no-cov', '-q', '--junitxml=' + str(xml)]
    with (root / f'{name}-{phase}.log').open('wb') as log:
        completed = subprocess.run(command, env=env, stdout=log, stderr=subprocess.STDOUT, timeout=60)
    cases = []
    for case in ET.parse(xml).iter('testcase'):
        cases.append({'node': case.attrib['classname'] + '::' + case.attrib['name'], 'status': 'failed' if case.find('failure') is not None else 'error' if case.find('error') is not None else 'skipped' if case.find('skipped') is not None else 'passed'})
    return {'command': command, 'exit': completed.returncode, 'cases': cases}

receipts = []
try:
    for name, path, scope, before, after, nodes in controls:
        text = originals[path].decode()
        if scope == 'endpoint':
            section = endpoint
        elif scope == 'append':
            section = text[text.index('def append_allowlist('):text.index('\n\ndef _candidates(')]
        elif scope == 'reader':
            section = text[text.index('def _artist_decisions('):text.index('\n\ndef load_allowlisted_artists(')]
        else:
            section = text
        assert section.count(before) == 1, (name, section.count(before))
        changed = section.replace(before, after, 1)
        mutation = text.replace(section, changed, 1).encode()
        assert mutation != originals[path]
        receipt = {'name': name, 'path': str(path), 'before': before, 'after': after, 'nodes': nodes, 'mutated_sha256': sha(mutation)}
        try:
            path.write_bytes(mutation)
            receipt['red'] = run(name, 'red', nodes)
        finally:
            path.write_bytes(originals[path])
            purge_bytecode()
        assert all(p.read_bytes() == data for p, data in originals.items())
        receipt['restored_sha256'] = {str(p): sha(p.read_bytes()) for p in (main, gates)}
        receipt['green'] = run(name, 'green', nodes)
        receipts.append(receipt)
        (root / 'mutations.json').write_text(json.dumps({'source_sha256': {str(p): sha(data) for p, data in originals.items()}, 'controls': receipts}, indent=2) + '\n')
        assert receipt['red']['exit'] == 1, receipt
        assert all(case['status'] == 'failed' for case in receipt['red']['cases']), receipt
        assert receipt['green']['exit'] == 0, receipt
        assert all(case['status'] == 'passed' for case in receipt['green']['cases']), receipt
        print(name, len(receipt['red']['cases']), 'failed -> passed; exits 1 -> 0; exact bytes restored', flush=True)
finally:
    for path, data in originals.items():
        path.write_bytes(data)
    purge_bytecode()
