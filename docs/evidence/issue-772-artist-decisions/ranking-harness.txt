import collections
import json
import re
from pathlib import Path

root = Path('/tmp/issue-772-round')
cov = json.loads((root / 'baseline-coverage.json').read_text())
counts = collections.defaultdict(lambda: {'repairs': 0, 'churn': 0})
repair = False
seen = set()
commits = []
for line in (root / 'history.txt').read_text().splitlines():
    if line.startswith('COMMIT '):
        sha, subject = line[7:].split(' ', 1)
        commits.append(sha)
        repair = bool(re.search(r'\b(fix|bug|correct|repair|guard|regression)\w*', subject, re.I))
        seen = set()
    elif line.startswith('src/fine_art_archive/') and line.endswith('.py') and line not in seen:
        counts[line]['churn'] += 1
        counts[line]['repairs'] += int(repair)
        seen.add(line)
rows = []
for path, data in cov['files'].items():
    s = data['summary']
    if path.startswith('src/fine_art_archive/') and (s['missing_lines'] or s['missing_branches']):
        rows.append({'path': path, **counts[path], **s})
rows.sort(key=lambda r: (-r['repairs'], -r['churn'], -r['missing_lines'], r['path']))
(root / 'ranking.json').write_text(json.dumps({'commits': len(commits), 'newest': commits[0], 'oldest': commits[-1], 'ranking': rows}, indent=2) + '\n')
print(json.dumps(cov['totals'], indent=2))
for r in rows[:6]:
    print(r['path'], r['repairs'], r['churn'], r['missing_lines'], r['missing_branches'])
f = cov['files']['src/fine_art_archive/api/main.py']['functions']
for name in ['artist_decision', 'review_gate', '_write_sidecar_atomic']:
    print(name, f[name]['summary'], f[name]['missing_lines'], f[name]['missing_branches'])
